find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
foreach(test IN ITEMS controller)
    add_executable(tbempty_test_${test} tst_${test}.cpp)
    set_target_properties(tbempty_test_${test} PROPERTIES AUTOMOC ON)
    if(test STREQUAL "controller" OR test STREQUAL "states")
        target_link_libraries(tbempty_test_${test} PRIVATE TBScrollViewEmpty::Core Qt${QT_VERSION_MAJOR}::Test)
    else()
        target_link_libraries(tbempty_test_${test} PRIVATE TBScrollViewEmpty::Widgets Qt${QT_VERSION_MAJOR}::Test)
    endif()
    add_test(NAME tbempty.${test} COMMAND tbempty_test_${test})
endforeach()
